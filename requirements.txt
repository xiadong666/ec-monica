# EC-Monica 依赖清单（Python 3.11）

# 版本说明：langchain / langgraph 的版本组合沿用已被验证过的搭配

# --- Agent / LLM 框架 ---
langchain==1.2.18          # create_agent 统一入口、@tool 装饰器
langchain-openai==1.6.1    # OpenAI 兼容客户端（DeepSeek 走这个接口）
langgraph==1.1.10          # 图编排运行时（对照实现，生产走显式控制流）

# --- Web 服务 ---
fastapi==0.141.1
uvicorn==0.52.4

# --- 配置 ---
python-dotenv==1.2.3

# --- 测试与评测 ---
# ★ 必须写进依赖清单：CI 只装 requirements.txt，
#   如果漏了 pytest，CI 的评测步骤会直接 `pytest: command not found`（退出码 127）。
#   踩过的坑：本地是手动 pip install pytest 的，所以本地全绿、CI 直接失败 ——
#   "在我机器上能跑" 的典型成因。
pytest==9.1.1

# --- 数据库驱动 ---
# PostgreSQL 后端需要。SQLite 走标准库 sqlite3，不需要额外依赖。
# 用 [binary] 额外装预编译包，避免 CI / 容器里还要装 libpq 开发头文件。
psycopg[binary]==3.3.6

# --- 指标层 ---
# 指标计算刻意只用标准库 sqlite3 + 纯 Python，
# 避免 pandas 版本差异导致口径漂移（可复现优先）。
#
# --- 安全与可观测 ---
# 认证（JWT/PBKDF2）、限流、结构化日志、审计**全部用标准库实现**，
# 不引入额外依赖 —— 这样项目可以零障碍运行，也减少供应链风险。
# 代价是：生产环境建议替换为成熟库（见 README 的进阶建议）。
#
# --- 可选：多副本部署 ---
# 会话外置到 Redis 时需要。单实例运行（EC_SESSION_BACKEND=memory）不需要。
# redis>=5.0.0
